# Issue #327 — DB Indexes: Eliminate Silent O(n) Lookups

> **For Pi:** Use `executing-plans` to implement this plan task-by-task.

**Goal:** Ensure every entity-resolution query executes via FalkorDB range indexes (Node By Index Scan) instead of All Node Scan, and prove it with schema-level + plan-shape tests.

**Team:** epistemic-team
**Role:** worker agent (issue #327)

**Architecture:** FalkorDB (embedded falkordblite in tests, server in prod) exposes single-property RANGE indexes per label. Indexes apply only when the query pattern carries the label AND filters the indexed property (empirically verified via `g.explain()`: labeled -> `Node By Index Scan`; unlabeled / label-in-WHERE / OR-across-unindexed -> `All Node Scan` / `Label Scan`; OR across TWO indexed props on one label -> `Node By Index Scan`). Existing `FalkorProjection._ensure_indexes()` already creates Point(id, pointKind, content_hash, is_operator) + Document(id, documentKind). Missing: (a) indexes on other entity canonical keys, (b) a `_resolve_entity` union helper, (c) rewriting unlabeled hot-path queries (sdk entity CRUD, projection edges, navigation BFS, org queries) to labeled index-backed patterns, (d) plan-shape + behavioral tests.

### Pattern Research

Skipped external research intake — plan touches zero new third-party dependencies. FalkorDB index behavior researched empirically during scoping (controller + 3 verifier cycles verified against `g.explain()` on falkordblite embedded) and confirmed against docs.falkordb.com/cypher/indexing (single-property range indexes; `CREATE INDEX FOR (n:Label) ON (n.prop)`; `CALL db.indexes()` lists them; no `IF NOT EXISTS` — "already indexed" error is the idempotency signal).

**Step 3: Implement**

```python
    _RESOLVE_BRANCHES = (
        # (label, key_property) — every branch is backed by a RANGE index
        # created in _ensure_indexes (issue #327). Event matches by eventId
        # (Event.id == eventId for all current writes; eventId-only legacy
        # nodes covered). Source matches by id and/or url (url-only
        # ingestion stubs have no id — _link_source).
        ("Point", "id"), ("Subject", "id"), ("Object", "id"),
        ("Document", "id"), ("Source", "id"),
        ("Event", "eventId"), ("Source", "url"),
    )

    def _resolve_entity(self, id_val: str, *, by_id: bool = True,
                        by_eventId: bool = False, by_url: bool = False) -> list[dict]:
        """Index-backed entity resolution mirroring the legacy
        `MATCH (n) WHERE n.id=$id OR n.eventId=$id OR n.url=$id` semantics.

        Returns [{label, key, value, properties}, ...] — one entry per matching
        node, deduped by node identity (a Source with id==url matches two
        branches). Each branch is a labeled lookup on an indexed property, so
        the planner uses Node By Index Scan instead of All Node Scan (#327).
        """
        branches = []
        for label, prop in self._RESOLVE_BRANCHES:
            if prop == "id" and not by_id:
                continue
            if prop == "eventId" and not by_eventId:
                continue
            if prop == "url" and not by_url:
                continue
            branches.append((label, prop))
        if not branches:
            return []
        union_q = " UNION ".join(
            f"MATCH (n:{label}) WHERE n.{prop} = $id "
            f"RETURN '{label}' AS label, '{prop}' AS key, n.{prop} AS value, "
            f"properties(n) AS props LIMIT 1"
            for label, prop in branches
        )
        rows = self.g.query(union_q, params={"id": id_val}).result_set
        seen: set[str] = set()
        out: list[dict] = []
        for label, key, value, props in rows:
            ident = props.get("id") or props.get("eventId") or props.get("url")
            if ident in seen:
                continue
            seen.add(ident)
            out.append({"label": label, "key": key, "value": value,
                        "properties": dict(props)})
        return out
```

```python
    def _get_entity(self, id_val: str) -> dict:
        # NOTE: Session/APIKey/Team/Tag nodes are intentionally excluded from
        # entity resolution (issue #327) - only Point/Subject/Object/Document/
        # Event/Source resolve. On a cross-label id collision the first
        # _RESOLVE_BRANCHES match wins (Point priority) - more deterministic
        # than the previous scan-order-arbitrary LIMIT 1.
        resolved = self._get_proj()._resolve_entity(
            id_val, by_id=True, by_eventId=True, by_url=True)
        return resolved[0]["properties"] if resolved else {}

    def _update_entity(self, id_val: str, **props) -> dict:
        proj = self._get_proj()
        # Per-label indexed writes (id OR eventId — original predicate).
        # UNION cannot carry SET, so run each branch sequentially.
        for label, prop in (("Point", "id"), ("Subject", "id"), ("Object", "id"),
                            ("Document", "id"), ("Source", "id"), ("Event", "eventId")):
            proj.g.query(
                f"MATCH (n:{label} {{{prop}:$id}}) SET n += $p",
                params={"id": id_val, "p": props},
            )
        return self._get_entity(id_val)

    def _delete_entity(self, id_val: str) -> bool:
        proj = self._get_proj()
        total = 0
        for label, prop in (("Point", "id"), ("Subject", "id"), ("Object", "id"),
                            ("Document", "id"), ("Source", "id"), ("Event", "eventId")):
            r = proj.g.query(
                f"MATCH (n:{label} {{{prop}:$id}}) DETACH DELETE n RETURN count(n)",
                params={"id": id_val},
            )
            if r.result_set:
                total += r.result_set[0][0]
        return bool(total)
```

```python
        """Create a named edge between two entities by their IDs.  (docstring kept)"""
```

```python
    def get_owned_entities(self, subject_id: str) -> list:
        proj = self._get_proj()
        r = proj.g.query(
            "MATCH (s:Subject) WHERE s.id = $sid OR s.name = $sid "
            "MATCH (s)<-[:ownedBy]-(e) RETURN properties(e) LIMIT 100",
            params={"sid": subject_id},
        )
        return [dict(row[0]) for row in r.result_set]

    def get_org_structure(self, subject_id: str) -> dict:
        proj = self._get_proj()
        members = proj.g.query(
            "MATCH (s:Subject) WHERE s.id = $sid OR s.name = $sid "
            "MATCH (p:Subject)-[:memberOf]->(s) RETURN properties(p)",
            params={"sid": subject_id},
        )
        roles = proj.g.query(
            "MATCH (p:Subject) WHERE p.id = $sid OR p.name = $sid "
            "MATCH (p)-[:holdsRole]->(r:Subject) RETURN properties(r)",
            params={"sid": subject_id},
        )
        return {
            "members": [dict(row[0]) for row in members.result_set],
            "roles": [dict(row[0]) for row in roles.result_set],
        }
```
